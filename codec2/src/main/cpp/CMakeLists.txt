cmake_minimum_required(VERSION 3.22)
project(meshstar_codec2 C)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
add_compile_options("-ffile-prefix-map=${REPO_ROOT}=." "-fdebug-prefix-map=${REPO_ROOT}=.")
set(NATIVE "${CMAKE_CURRENT_LIST_DIR}/../../../build/native")
set(SOURCE "${NATIVE}/source/src")
set(GENERATED "${NATIVE}/generated")
set(NAMES dump lpc nlp postfilter sine codec2 codec2_fft kiss_fft kiss_fftr interp lsp mbest newamp1 phase quantise pack)
foreach(NAME ${NAMES})
    list(APPEND SPEECH "${SOURCE}/${NAME}.c")
endforeach()
file(GLOB TABLES "${GENERATED}/codebook*.c")
add_library(codec2 SHARED ${SPEECH} ${TABLES})
target_include_directories(codec2 PUBLIC "${SOURCE}" "${GENERATED}")
target_compile_options(codec2 PRIVATE -O2 -ffp-contract=off)
target_link_libraries(codec2 PRIVATE m)
add_library(meshstar_codec2_jni SHARED bridge.c)
target_link_libraries(meshstar_codec2_jni PRIVATE codec2)
